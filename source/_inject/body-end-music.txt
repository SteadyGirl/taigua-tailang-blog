<!-- 音频播放器（隐藏） -->
<audio id="audioPlayer" preload="none">

<script>
// ==========【这里改你的歌曲！！】==========
// 格式：["文件名.mp3","歌曲显示名字"]
const songList = [
  {src:"/music/椎名林檎 - 幸福論.mp3", name:"幸福論 - 椎名林檎"},
  {src:"/music/Nujabes,Fat Jon - Battlecry.mp3", name:"Battlecry - Nujabes,Fat Jon"},
  {src:"/music/Just the two of us - Bill Withers.mp3", name:"Just the two of us"}
];
// ======================================

const btn = document.getElementById('musicBtn');
const panel = document.getElementById('musicPanel');
const listWrap = document.getElementById('musicList');
const audio = document.getElementById('audioPlayer');
let currentIndex = 0;

// 渲染歌单
function renderList(){
  listWrap.innerHTML = '';
  songList.forEach((item,idx)=>{
    const div = document.createElement('div');
    div.className = "music-item";
    div.innerText = item.name;
    div.onclick = ()=> playSong(idx);
    listWrap.appendChild(div);
  })
}

// 播放指定序号歌曲
function playSong(idx){
  currentIndex = idx;
  audio.src = songList[idx].src;
  audio.play();
}

// 自动下一首（连续播放核心）
audio.addEventListener('ended',()=>{
  currentIndex++;
  if(currentIndex < songList.length){
    playSong(currentIndex);
  }else{
    // 播放完最后一首，停止
    audio.pause();
  }
})

// 点击按钮 显示/隐藏歌单弹窗
btn.onclick = ()=>{
  if(panel.style.display === 'block'){
    panel.style.display = 'none';
  }else{
    panel.style.display = 'block';
  }
}

renderList();
</script>
